Aim and Essential Theory

Aim: To add, update, delete (drop), and view student records using an HTML frontend, a Java backend, and JDBC connectivity to an Oracle Database.

The database contains a STUDENTS table with attributes: roll_no (Primary Key), name, email (Unique), department, semester, and gpa. User actions in the web frontend dispatch HTTP requests to the Java backend. The backend executes pre-compiled SQL queries via PreparedStatement to perform INSERT, UPDATE, and DELETE operations.

Connection String: jdbc:oracle:thin:@localhost:1521:xe | User: system | Driver: oracle.jdbc.OracleDriver (Type 4 Thin Driver). executeUpdate() returns the number of rows modified. Fresh SELECT queries retrieve verified records.

Run

Install JDK 17 or newer. Open project directory and launch:

  • Windows: Double click run.bat
  • Linux / macOS: Run sh run.sh

Then open http://localhost:5000 in your web browser. Stop the server with Ctrl + C.

Complete Backend Source Code

File: StudentDAO.java

package com.student.dao;

import com.student.model.Student;
import com.student.util.DBConnection;
import java.sql.*;
import java.util.*;

public class StudentDAO {

    public List<Student> getAllStudents() throws SQLException {
        List<Student> list = new ArrayList<>();
        String sql = "SELECT roll_no, name, email, department, semester, gpa FROM students ORDER BY roll_no ASC";
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql);
             ResultSet rs = pstmt.executeQuery()) {
            while (rs.next()) {
                list.add(new Student(
                    rs.getString("roll_no"),
                    rs.getString("name"),
                    rs.getString("email"),
                    rs.getString("department"),
                    rs.getInt("semester"),
                    rs.getDouble("gpa")
                ));
            }
        }
        return list;
    }

    public boolean addStudent(Student s) throws SQLException {
        String sql = "INSERT INTO students (roll_no, name, email, department, semester, gpa) VALUES (?, ?, ?, ?, ?, ?)";
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql)) {
            pstmt.setString(1, s.getRollNo().trim());
            pstmt.setString(2, s.getName().trim());
            pstmt.setString(3, s.getEmail().trim());
            pstmt.setString(4, s.getDepartment().trim());
            pstmt.setInt(5, s.getSemester());
            pstmt.setDouble(6, s.getGpa());
            return pstmt.executeUpdate() > 0;
        }
    }

    public boolean updateStudent(Student s) throws SQLException {
        String sql = "UPDATE students SET name = ?, email = ?, department = ?, semester = ?, gpa = ? WHERE roll_no = ?";
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql)) {
            pstmt.setString(1, s.getName().trim());
            pstmt.setString(2, s.getEmail().trim());
            pstmt.setString(3, s.getDepartment().trim());
            pstmt.setInt(4, s.getSemester());
            pstmt.setDouble(5, s.getGpa());
            pstmt.setString(6, s.getRollNo().trim());
            return pstmt.executeUpdate() > 0;
        }
    }

    public boolean dropStudent(String rollNo) throws SQLException {
        String sql = "DELETE FROM students WHERE roll_no = ?";
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql)) {
            pstmt.setString(1, rollNo.trim());
            return pstmt.executeUpdate() > 0;
        }
    }
}

Complete Database Schema & Launcher Code

File: schema.sql

CREATE TABLE students (
    roll_no     VARCHAR2(20)    NOT NULL PRIMARY KEY,
    name        VARCHAR2(100)   NOT NULL,
    email       VARCHAR2(100)   NOT NULL UNIQUE,
    department  VARCHAR2(50)    NOT NULL,
    semester    NUMBER(2)       DEFAULT 1,
    gpa         NUMBER(3,2)     DEFAULT 0.00
);
COMMIT;
